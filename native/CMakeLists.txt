cmake_minimum_required(VERSION 3.16)
project(nashtree_inference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NASHTREE_SANITIZE "Enable address/undefined sanitizers" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
  if(NASHTREE_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(nashtree_inference STATIC ../src/inference.cpp)
target_include_directories(nashtree_inference PUBLIC ../src/include)
add_executable(ntf-convert convert.cpp)
target_link_libraries(ntf-convert PRIVATE nashtree_inference)
add_executable(ntf-predict cli.cpp)
target_link_libraries(ntf-predict PRIVATE nashtree_inference)
add_executable(ntf-test test.cpp)
target_link_libraries(ntf-test PRIVATE nashtree_inference)
enable_testing()
add_test(NAME native-contract COMMAND ntf-test)
if(UNIX)
  add_executable(ntf-lifetime lifetime.cpp)
  target_link_libraries(ntf-lifetime PRIVATE nashtree_inference)
endif()
